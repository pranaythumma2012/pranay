#include <bits/stdc++.h>
using namespace std;

#define ll long long int

int main() 
{
    ll t;
    cin >>t;
    while(t--)
    {
        ll n;
        cin >>n;
        
        if(n<4||n%2!=0)
        {
            cout<<-1<<endl;
        }
        else if(n==4||n==6)
        {
            cout <<"1 1"<<endl;
        }
        else
        {
            n=n/2;
            cout <<(n+2)/3<<" "<<n/2<<endl;
        }

    }
}

#include <bits/stdc++.h>
using namespace std;

int main() 
{
    int t;
    cin >>t;
    while(t--)
    {
        int n;
        cin >>n;
        int m=INT_MAX;
        int p=INT_MIN;
        vector<int>v(n);
        for(int i=0;i<n;i++)
        {
            cin >>v[i];
        }
        for(int i=0;i<n;i++)
        {
            m=min(m,v[i]);
            p=max(p,v[i]);
        }
        int hi=INT_MIN;
        hi =max(hi,p-v[0]);
        hi =max(hi,v[n-1]-m);
        for (int i=1;i<n;i++)
        {
            hi=max(hi,v[i-1]-v[i]);
        }
        hi=max(hi,v[n-1]-v[0]);
            cout <<hi<<endl;
    }
}
